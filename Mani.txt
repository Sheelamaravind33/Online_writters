const TYPE_BASE = {lab:300,assign:200,notes:150,project:500,research:700,thesis:1200,ppt:250,code:400};
const DEADLINE_LABELS = ['','24 hrs','48 hrs','3 days','5 days','7 days'];

function calcPrice() {
  const type = document.getElementById('estType').value;
  const complex = parseInt(document.getElementById('estComplex').value);
  const deadline = parseInt(document.getElementById('estDeadline').value);

  document.getElementById('complexLabel').textContent = COMPLEX_LABELS[complex];
  document.getElementById('deadlineLabel').textContent = DEADLINE_LABELS[deadline];

  const cRange = document.getElementById('estComplex');
  cRange.style.setProperty('--pct', `${((complex-1)/2)*100}%`);
  const dRange = document.getElementById('estDeadline');
  dRange.style.setProperty('--pct', `${((deadline-1)/4)*100}%`);

  const base = TYPE_BASE[type] || 300;
  const complexFee = (complex - 1) * 80;
  const rush = deadline === 1 ? Math.round(base * 0.6) : deadline === 2 ? Math.round(base * 0.3) : 0;
  const total = base + complexFee + rush;

  document.getElementById('estBase').textContent = `₹${base}`;
  document.getElementById('estRush').textContent = `₹${rush}`;
  document.getElementById('estComplexFee').textContent = `₹${complexFee}`;
  document.getElementById('estTotal').textContent = `₹${total}`;
}

function doLogin() {

  // Demo credentials OR registered users
  const users = DB.get('users') || [];

function doRegister() {
  const name = document.getElementById('regName').value.trim();
  const email = document.getElementById('regEmail').value.trim();
  const pass = document.getElementById('regPass').value;
  const err = document.getElementById('regError');

  if (!name || !email || !pass) { err.style.display = 'block'; err.textContent = 'Please fill all fields'; return; }
  if (pass.length < 6) { err.style.display = 'block'; err.textContent = 'Password must be at least 6 characters'; return; }

  const users = DB.get('users') || [];
  if (users.find(u => u.email === email)) { err.style.display = 'block'; err.textContent = 'Email already registered'; return; }

  err.style.display = 'none';
  DB.push('users', { name, email, pass, joinedOn: new Date().toISOString() });
  DB.set('currentUser', { email, name });
  setUserUI(name);
  notify('Account created! Welcome to Online Writers');
  showHome();
}

function setUserUI(name) {
  const initials = name.split(' ').map(w=>w[0]).join('').slice(0,2).toUpperCase();
  ['userAvatar','writerDetailAvatar','allWritersAvatar','bookingsAvatar'].forEach(id => {
    const el = document.getElementById(id);
    if (el) el.textContent = initials;
  });
}

function showRegister() {
  document.getElementById('loginBox').style.display = 'none';
  document.getElementById('registerBox').style.display = 'block';
}

function showLogin() {
  document.getElementById('registerBox').style.display = 'none';
  document.getElementById('loginBox').style.display = 'block';
}

// ─── NAVIGATION ───────────────────────────────────────────────────────────
function showScreen(id) {
  document.querySelectorAll('.screen').forEach(s => s.classList.remove('active'));
  const el = document.getElementById(id);
  if (el) { el.classList.add('active'); window.scrollTo(0,0); }
}

function showHome() {
  renderWorkGrid();
  renderFeaturedWriters();
  calcPrice();
  showScreen('home');
}

function showFindWriters() {
  renderAllWriters();
  showScreen('allWriters');
}

function showBookings() {
  renderBookings();
  showScreen('bookings');
}

function doSearch() {
  showFindWriters();
}

function closeModal() {
  document.getElementById('successModal').classList.remove('open');
}

function notify(msg) {
  const n = document.getElementById('notif');
  document.getElementById('notifText').textContent = msg;
  n.classList.add('show');
  setTimeout(() => n.classList.remove('show'), 3000);
}

// ─── INIT ──────────────────────────────────────────────────────────────
window.addEventListener('load', () => {
  const user = DB.get('currentUser');
  if (user) setUserUI(user.name);

  setTimeout(() => {
    showScreen('login');
  }, 3800);
});
